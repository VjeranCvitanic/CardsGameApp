cmake_minimum_required(VERSION 3.16)
project(CardsGameEngine LANGUAGES CXX)

# =========================
# Global settings
# =========================
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(BUILD_SHARED_LIBS "Build shared libraries" ON)
# =========================
# Common library
# =========================
add_library(card_engine STATIC
    # Engine base
    src/EngineBase/Cards.cpp
    src/EngineBase/Deck.cpp
    src/EngineBase/CardsRound.cpp
    src/EngineBase/CardsGame.cpp
    src/EngineBase/CardsMatch.cpp

    # Briscola
    src/Briscola/BriscolaRound.cpp
    src/Briscola/BriscolaGame.cpp
    src/Briscola/BriscolaMatch.cpp
    src/Briscola/BriscolaRules.cpp

    # Tressette
    src/Tressette/TressetteMatch.cpp
    src/Tressette/TressetteGame.cpp
    src/Tressette/TressetteRound.cpp
    src/Tressette/TressetteRules.cpp

    # Events / infra
    src/Logger.cpp
    src/Acussos.cpp

    # Engine interface
    src/EngineIF.cpp
)

target_compile_features(card_engine PUBLIC cxx_std_20)

target_include_directories(card_engine
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/inc
        ${CMAKE_CURRENT_SOURCE_DIR}/inc/EngineBase
        ${CMAKE_CURRENT_SOURCE_DIR}/inc/Briscola
        ${CMAKE_CURRENT_SOURCE_DIR}/inc/Tressette
        ${CMAKE_CURRENT_SOURCE_DIR}/inc/Events
)

if(MSVC)
    target_compile_options(card_engine PRIVATE /W4 /permissive-)
else()
    target_compile_options(card_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
